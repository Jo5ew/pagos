const SHEET_NAME = 'Registros'; const SECRET_CODE = 'ANIV26'; const CUTOFF = '15:00'; // debe coincidir con HORA_LIMITE en el index.html const DRIVE_FOLDER_NAME = 'EvidenciasAsistencia'; function doPost(e) { try { const params = e.parameter; if (params.codigo !== SECRET_CODE) { return respond({ ok: false, mensaje: 'Código incorrecto' }); } const nombre = params.nombre; if (!nombre) { return respond({ ok: false, mensaje: 'Falta el nombre' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); const tz = Session.getScriptTimeZone(); const ahora = new Date(); const hoyStr = Utilities.formatDate(ahora, tz, 'yyyy-MM-dd'); const data = sheet.getDataRange().getValues(); let tieneEntrada = false; let tieneSalida = false; for (let i = 1; i < data.length; i++) { const fila = data[i]; if (!fila[2]) continue; const filaFecha = Utilities.formatDate(new Date(fila[2]), tz, 'yyyy-MM-dd'); if (fila[0] === nombre && filaFecha === hoyStr) { if (fila[1] === 'Entrada') tieneEntrada = true; if (fila[1] === 'Salida') tieneSalida = true; } } let tipo = 'Entrada'; if (tieneEntrada && !tieneSalida) tipo = 'Salida'; if (tieneEntrada && tieneSalida) { return respond({ ok: false, mensaje: 'Ya registraste tu entrada y salida hoy' }); } let fotoUrl = params.foto ? guardarFoto(params.foto, nombre, tipo) : ''; let fotoJust1Url = params.foto2 ? guardarFoto(params.foto2, nombre, tipo + '_justif1') : ''; let fotoJust2Url = params.foto3 ? guardarFoto(params.foto3, nombre, tipo + '_justif2') : ''; let tardanzaMin = ''; let estado = ''; if (tipo === 'Entrada') { const partesHora = CUTOFF.split(':').map(Number); const limite = new Date(ahora); limite.setHours(partesHora[0], partesHora[1], 0, 0); const diffMs = ahora - limite; tardanzaMin = diffMs > 0 ? Math.round(diffMs / 60000) : 0; estado = diffMs > 0 ? 'Tardanza' : 'A tiempo'; } else { estado = 'Salida registrada'; } sheet.appendRow([ nombre, tipo, ahora, tardanzaMin, estado, params.lat || '', params.lng || '', fotoUrl, params.comentario || '', fotoJust1Url, fotoJust2Url ]); return respond({ ok: true, tipo: tipo, tardanza: tardanzaMin }); } catch (err) { return respond({ ok: false, mensaje: 'Error: ' + err.message }); } } function guardarFoto(base64Data, nombre, tipo) { const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME); const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME); const partes = base64Data.split(','); const bytes = Utilities.base64Decode(partes[1] || partes[0]); const blob = Utilities.newBlob(bytes, 'image/jpeg', nombre + '_' + tipo + '_' + Date.now() + '.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const nombres = ss.getSheets().map(s => s.getName()); return ContentService.createTextOutput(JSON.stringify({ archivo: ss.getName(), pestañas: nombres })).setMimeType(ContentService.MimeType.JSON); } function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
